# How to start a project in C#?

First, use this command to create a new repo project:

`dotnet new console -n NomeDoProjeto`

Then go to your project repo in the terminal and run `dotnet run` to execute the project.

# Numeric Data Types.

```C#
using System;

namespace CSharp
{
	internal class Program
	{
		static void Main(string[] args)
		{
			int age = 23;

			Console.WriteLine(age);
			
			Console.WriteLine(int.MaxValue);
			
			Console.WriteLine(int.MinValue);
			
			  
			
			long bigNumber = 90000000L;
			
			Console.WriteLine(bigNumber);
			
			Console.WriteLine(long.MaxValue);
			
			Console.WriteLine(long.MinValue);
			
			  
			
			double negative = -55.2D;
			
			Console.WriteLine(negative);
			
			Console.WriteLine(double.MaxValue);
			
			Console.WriteLine(double.MinValue);
			
			  
			
			float precision = 5.00001F;
			
			Console.WriteLine(precision);
			
			Console.WriteLine(float.MaxValue);
			
			Console.WriteLine(float.MinValue);
			
			  
			
			decimal money = 14.00M;
			
			Console.WriteLine(money);
			
			Console.WriteLine(decimal.MaxValue);
			
			Console.WriteLine(decimal.MinValue);
			
			  
			  
			
			Console.ReadLine();
		}
	}
}
```

# Text based Data Type

```C#
using System;

namespace CSharp
{
	internal class Program
	{
		static void Main(string[] args)
		{
			string name = "Eduardo";

			char letter = 'a';
			
			Console.WriteLine(name + "\n" + letter);
			
			  
			
			Console.Write("Your Name Is: ");
			
			Console.Write(name);
			
			Console.ReadLine();
		}
	}
}
```
